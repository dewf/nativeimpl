// OPENWL ======================================================================================

flags Modifiers {
    Shift = 1 << 0,
    Control = 1 << 1,
    Alt = 1 << 2,
    MacControl = 1 << 3
}

enum MouseEventType {
    Down,
    Up,
    Move,
    Enter,
    Leave,
    Wheel
}

enum MouseButton {
    None,
    Left,
    Middle,
    Right,
    Other
}

enum MouseWheelAxis {
    None,
    Vertical,
    Horizontal
}

struct CloseRequestReply {
    bool cancelClose;
}

struct MouseEvent {
    MouseEventType eventType;
    int x;
    int y;
    int wheelDelta;
    MouseWheelAxis wheelAxis;
    MouseButton button;
    Modifiers modifiers;
}

interface WindowDelegate {
    CloseRequestReply onCloseRequest();
    void onDestroyed();
    void onResized(int newWidth, int newHeight);
    void onRepaint(CGContext context);
    void onAction(int id, Action action);
    void onMouseEvent(MouseEvent mouseEvent);

    // drop stuff
    DropEffect onDropFeedback(DropData dropData, DropEffect defaultAction, int x, int y, Modifiers modifiers);
    void onDropFinal(DropData dropData, DropEffect dropAction, int x, int y, Modifiers modifiers);
    void onDropLeave();
}

opaque Window {
    void destroy();
    void show();
    void showRelative(Window relativeTo, int x, int y, int newWidth, int newHeight);
    void showModal(Window parent); // null for app-modal
    void endModal();
    void hide();
    void invalidate(int x, int y, int width, int height); // or 0s for whole window
    size_t getOSHandle();

    void setFocus();
    void mouseGrab();

    // misc involving other objects:
    void setMenuBar(MenuBar mb);
    void setCursor(Cursor cursor);
    void showContextMenu(Menu menu, int x, int y); // no 'fromEvent' for now, will try to get rid of that in the linux version
    void enableDrops(bool enabled);
}
Window createWindow(int width, int height, string title, WindowDelegate del);
void mouseUngrab();

    None = 0,
    Copy = 1 << 0,
    Move = 1 << 1,
    Link = 1 << 2,
    Other = 1 << 3
}

    void release();
    void addFormat(string dragFormatMIME);

    Bytes(Buffer<byte> data),
    Files(Array<string> filenames),
    UTF8(string value)
}

    bool hasFormat(string dropFormatMIME);
    DropDataFormat getUTF8();

// FILE DIALOG API ==========================

enum FileDialogMode {
    File,
    Folder
}

struct FileDialogFilterSpec {
    string description;
    Array<string> extensions;   // eg "*.jpg"
}

struct FileDialogOpts {
    Window forWindow;                       // optional, otherwise app-modal
    FileDialogMode mode;
    Array<FileDialogFilterSpec> filters;
    bool allowAll;                          // append *.* filter?
}

sumtype FileDialogResult {
    Success(Array<string> filenames),
    Canceled
}

FileDialogResult fileOpenDialog(FileDialogOpts opts, bool allowMultiple);
FileDialogResult fileSaveDialog(FileDialogOpts opts, string suggestedFilename, string defaultExt);
