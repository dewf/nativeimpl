module TestLibrary;

interface WindowDelegate {
    void onDestroyed();
    void onResized(int newWidth, int newHeight);
    void onRepaint(CGContext context);
    void onAction(int id, Action action);
}

opaque Window {
   void show();
   void setMenuBar(MenuBar mb);
   void invalidate(int x, int y, int width, int height); // or 0s for whole window
   void destroy();
}
Window createWindow(string title, WindowDelegate del);

opaque Accelerator;
Accelerator createAccelerator(KeyEnum key);                 // TODO: modifiers (bitmask / combinable set)

opaque Action;
Action createAction(int id, string label, Accelerator acc); // special nullable/optional syntax?

opaque Menu {
   void addAction(Action action);
   void addSubmenu(string label, Menu sub);
   void addSeparator();
}
Menu createMenu();

opaque MenuBar {
   void addMenu(string label, Menu menu);
}
MenuBar createMenuBar();

struct CGPoint {
   double x;
   double y;
}

struct CGSize {
   double width;
   double height;
}

struct CGRect {
   CGPoint origin;
   CGSize size;
}

opaque CGContext {
   void setRGBFillColor(double red, double green, double blue, double alpha);
   void fillRect(CGRect rect);
}

int runloop();
void exitRunloop();

int moduleInit();
void moduleShutdown();
